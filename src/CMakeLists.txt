cmake_minimum_required(VERSION 3.9)
project(smash_and_dash)

file(GLOB_RECURSE SOURCES
    src/*.h
    src/*.cpp
)

string(REGEX REPLACE "[^;]*CMakeFiles/[^;]+;?" "" SOURCES "${SOURCES}")

include_directories(src)

# The world generator (plain C++, no engine) : compiled once, for the game and for its test (sad_worldgen_test)
# (With the pure C++ of the chunks : their images, their entity records)
file(GLOB WORLDGEN_SOURCES CONFIGURE_DEPENDS src/Game/Gameplay/World/Generation/*.cpp src/Game/Gameplay/World/Chunks/*.cpp)
list(REMOVE_ITEM SOURCES ${WORLDGEN_SOURCES})
# The web worker of the chunks : a module of its own (below)
file(GLOB CHUNK_WORKER_SOURCES CONFIGURE_DEPENDS src/Game/Gameplay/World/ChunkWorker/*.cpp)
list(REMOVE_ITEM SOURCES ${CHUNK_WORKER_SOURCES})
# The web worker of the infinite world : a module of its own (below)
file(GLOB WORLD_WORKER_SOURCES CONFIGURE_DEPENDS src/Game/Gameplay/World/WorldWorker/*.cpp)
list(REMOVE_ITEM SOURCES ${WORLD_WORKER_SOURCES})
add_library(smash_and_dash_worldgen OBJECT ${WORLDGEN_SOURCES})
set_target_properties(smash_and_dash_worldgen PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Optimised in a Debug build too : a world takes about 75 s unoptimised (the painted tiles blended at 48 px), against
# 60 s for a run of the agent tools (tools/doctor, the Debug CI) to be ready ; its checks are the golden tests
if(NOT MSVC)
	target_compile_options(smash_and_dash_worldgen PRIVATE $<$<CONFIG:Debug>:-O2>)
endif()

# The sources of the game : compiled without waiting for the engine to be linked (the engine headers are on the include
# path of every target : the root CMakeLists.txt), the game library links them with the engine
add_library(smash_and_dash_objects OBJECT ${SOURCES})
set_target_properties(smash_and_dash_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)

# The game is a library ("libGame") loaded and run by the ska launcher executable (ska_main). On the web, it is linked
# with the engine in the page (ska_web_executable)
if(EMSCRIPTEN)
	add_library(smash_and_dash STATIC $<TARGET_OBJECTS:smash_and_dash_objects> $<TARGET_OBJECTS:smash_and_dash_worldgen>)
else()
	add_library(smash_and_dash SHARED $<TARGET_OBJECTS:smash_and_dash_objects> $<TARGET_OBJECTS:smash_and_dash_worldgen>)
endif()
set_target_properties(smash_and_dash PROPERTIES OUTPUT_NAME Game)
add_dependencies(smash_and_dash ska_main)
set(TARGET_LIBS skalog skabase ska_ai ska_audio ska_core ska_draw ska_gui ska_particle ska_physic_chipmunk ska_script ska_scripting ska_world)
target_link_libraries(smash_and_dash ${TARGET_LIBS})
ska_precompile_std(smash_and_dash_objects smash_and_dash_worldgen)
# The objects get the include directories the engine libraries give to the game (TARGET_PROPERTY : no build order)
foreach(objects smash_and_dash_objects smash_and_dash_worldgen)
	foreach(library ${TARGET_LIBS})
		target_include_directories(${objects} PRIVATE $<TARGET_PROPERTY:${library},INTERFACE_INCLUDE_DIRECTORIES>)
	endforeach()
endforeach()

# The game bindings script templates are deployed with the skalang standard library ones (in "std", next to the executables)
add_custom_command(TARGET smash_and_dash POST_BUILD
	COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/scripts ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/std
	COMMENT "Deploying the game script templates")

if(EMSCRIPTEN)
	# The page of the game : the game directories, the scripts of the standard library and of the bindings ("std", next
	# to the executable : /bin/std)
	ska_web_executable(smash_and_dash_web
		GAME smash_and_dash
		ASSETS_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..
		ASSETS Data Levels Sprites Sounds Music Fonts Menu Resources Shaders gamesettings.ini
		SHELL ${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html
		# The worlds are generated by the game (in the saves : a world left out is generated again from its seed)
		LINK_OPTIONS "--preload-file=${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/std@/bin/std" "--exclude-file=*/Data/Saves/*/World" "--exclude-file=*/Data/Worlds")
	set_target_properties(smash_and_dash_web PROPERTIES OUTPUT_NAME index)
	add_dependencies(smash_and_dash_web skalang_std_scripts)
	# chunk_worker.js, next to the page : it decodes the images of the chunks away from the game (ChunkPrepareQueue.cpp)
	add_executable(sad_chunk_worker ${CHUNK_WORKER_SOURCES} src/Game/Gameplay/World/Chunks/ChunkImages.cpp src/Game/Gameplay/World/Generation/Png.cpp)
	set_target_properties(sad_chunk_worker PROPERTIES OUTPUT_NAME chunk_worker SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY ${CMAKE_RUNTIME_OUTPUT_DIRECTORY})
	target_link_options(sad_chunk_worker PRIVATE -sBUILD_AS_WORKER -sALLOW_MEMORY_GROWTH -sEXPORTED_FUNCTIONS=_prepareChunk)
	add_dependencies(smash_and_dash_web sad_chunk_worker)
	# world_worker.js, next to the page : it writes the chunks of the infinite world away from the game (ChunkGenerator.cpp)
	file(GLOB WORLDGEN_GENERATION_SOURCES CONFIGURE_DEPENDS src/Game/Gameplay/World/Generation/*.cpp)
	add_executable(sad_world_worker ${WORLD_WORKER_SOURCES} ${WORLDGEN_GENERATION_SOURCES} src/Game/Gameplay/World/Chunks/FileBundle.cpp)
	set_target_properties(sad_world_worker PROPERTIES OUTPUT_NAME world_worker SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY ${CMAKE_RUNTIME_OUTPUT_DIRECTORY})
	target_compile_options(sad_world_worker PRIVATE -O2)
	target_link_options(sad_world_worker PRIVATE -O2 -sBUILD_AS_WORKER -sALLOW_MEMORY_GROWTH -sSTACK_SIZE=1MB -sFORCE_FILESYSTEM
		-sEXPORTED_FUNCTIONS=_setupWorld,_writeChunk)
	add_dependencies(smash_and_dash_web sad_world_worker)
	# The scripts are checked and tested by the desktop build
	return()
endif()

# Static check of the scripts (skalang_check_scripts) : the build fails when a script does not compile
# (syntax, types, calls of the engine and game functions, declared by their templates)
set(GAME_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/..)
# Scripts only imported by the other scripts (no "run" function)
set(GAME_SCRIPT_MODULES
	${GAME_ROOT_DIR}/Levels/dialog.miniska
	${GAME_ROOT_DIR}/Levels/surf.miniska
	${GAME_ROOT_DIR}/Levels/jump_wall.miniska
	${GAME_ROOT_DIR}/Levels/shop.miniska
	# Called by the battles (BattleRules)
	${GAME_ROOT_DIR}/Data/Battle/rules.miniska
	${GAME_ROOT_DIR}/Data/Battle/ai.miniska
	${GAME_ROOT_DIR}/Data/Battle/status.miniska
	${GAME_ROOT_DIR}/Data/Battle/stages.miniska
	${GAME_ROOT_DIR}/Data/Battle/encounters.miniska
	${GAME_ROOT_DIR}/Data/Battle/skills.miniska
	${GAME_ROOT_DIR}/Data/Battle/rewards.miniska
	${GAME_ROOT_DIR}/Data/Battle/lines.miniska
	${GAME_ROOT_DIR}/Data/Battle/bond.miniska
	${GAME_ROOT_DIR}/Data/Battle/clash.miniska
	${GAME_ROOT_DIR}/Data/Battle/dash.miniska)
file(GLOB_RECURSE GAME_SCRIPTS CONFIGURE_DEPENDS ${GAME_ROOT_DIR}/Levels/*.miniska ${GAME_ROOT_DIR}/Resources/*.miniska)
# The agent scripts (run by tools/ska-console : tools/agent of the game and of the engine)
file(GLOB AGENT_SCRIPTS CONFIGURE_DEPENDS ${GAME_ROOT_DIR}/tools/agent/*.miniska ${GAME_ROOT_DIR}/external/ska/tools/agent/*.miniska)
list(APPEND GAME_SCRIPTS ${AGENT_SCRIPTS})
# The steps of the battles : a script started by the battle (BattleDirectorSystem)
list(APPEND GAME_SCRIPTS ${GAME_ROOT_DIR}/Data/Battle/steps.miniska)
# The arrivals of the wild Pokemon flying in or out of a hideout : a script started by the spawner (WorldState)
list(APPEND GAME_SCRIPTS ${GAME_ROOT_DIR}/Data/Battle/arrivals.miniska)
list(REMOVE_ITEM GAME_SCRIPTS ${GAME_SCRIPT_MODULES})
skalang_check_scripts(smash_and_dash_scripts_check
	SCRIPTS ${GAME_SCRIPTS}
	MODULES ${GAME_SCRIPT_MODULES}
	BIND_DIRECTORIES ${SKA_SCRIPT_TEMPLATES_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/scripts
	WORKING_DIRECTORY ${GAME_ROOT_DIR})

# The C++ port of the world generator (World/Generation : plain C++, no engine) against the values of the Python
# generator (tests/worldgen/golden.txt, tools/worldgen/golden.py)
add_executable(sad_worldgen_test ${GAME_ROOT_DIR}/tests/worldgen/WorldgenTest.cpp ${GAME_ROOT_DIR}/tests/worldgen/InfiniteTest.cpp $<TARGET_OBJECTS:smash_and_dash_worldgen>)
target_include_directories(sad_worldgen_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${GAME_ROOT_DIR}/external/ska/external/doctest)
add_test(NAME sad_worldgen COMMAND sad_worldgen_test WORKING_DIRECTORY ${GAME_ROOT_DIR})

# The pure C++ of the chunks of the world (World/Chunks)
add_executable(sad_chunks_test ${GAME_ROOT_DIR}/tests/chunks/ChunksTest.cpp $<TARGET_OBJECTS:smash_and_dash_worldgen>)
target_include_directories(sad_chunks_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${GAME_ROOT_DIR}/external/ska/external/doctest)
find_package(Threads REQUIRED)
target_link_libraries(sad_chunks_test Threads::Threads)
add_test(NAME sad_chunks COMMAND sad_chunks_test WORKING_DIRECTORY ${GAME_ROOT_DIR})

# Headless script tests (ctest, no window) : the scripts run with the templates of the engine and game functions (their
# default bodies), and the fakes of tests/fakes (ex : an engine keeping the game variables)
file(GLOB GAME_SCRIPT_TESTS CONFIGURE_DEPENDS ${GAME_ROOT_DIR}/tests/*_test.miniska)
skalang_test_scripts(game_script
	TESTS ${GAME_SCRIPT_TESTS}
	BIND_DIRECTORIES ${SKA_SCRIPT_TEMPLATES_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/scripts ${GAME_ROOT_DIR}/tests/fakes
	WORKING_DIRECTORY ${GAME_ROOT_DIR})
